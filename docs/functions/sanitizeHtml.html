      var DEFAULT_ALLOWED_TAGS = [
        "b",
        "i",
        "em",
        "strong",
        "a",
        "ul",
        "ol",
        "li",
        "p",
        "br"
      ];
      var DEFAULT_ALLOWED_ATTRIBUTES = {
        a: ["href", "target", "rel"]
      };
      function sanitizeHtml(dirtyHtml, allowedTags = DEFAULT_ALLOWED_TAGS, allowedAttributes = DEFAULT_ALLOWED_ATTRIBUTES) {
        const parser = new DOMParser();
        const doc = parser.parseFromString(dirtyHtml, "text/html");
        const isDangerousAttr = (attrName, value) => {
          if (attrName === "href" || attrName === "src") {
            const val = value.trim().toLowerCase();
            return val.startsWith("javascript:") || val.startsWith("data:") || val.startsWith("vbscript:");
          }
          return false;
        };
        const sanitizeNode = (node) => {
          const tag = node.tagName.toLowerCase();
          if (!allowedTags.includes(tag)) {
            node.replaceWith(...Array.from(node.childNodes));
            return;
          }
          for (const attr of Array.from(node.attributes)) {
            const attrName = attr.name.toLowerCase();
            const tagAttrs = allowedAttributes[tag] || [];
            if (!tagAttrs.includes(attrName) || isDangerousAttr(attrName, attr.value)) {
              node.removeAttribute(attr.name);
            }
          }
        };
        const walker = doc.body.querySelectorAll("*");
        for (const el of walker) {
          sanitizeNode(el);
        }
        return doc.body.innerHTML;
      }
      export {
        sanitizeHtml
      };
