    DistanceUnit: "meters" | "kilometers" | "miles" | "yards"

    Unit of measurement for distance.

    Remarks

    This type defines the units that can be used for distance calculations in the haversineDistance function.

    // src/math/haversineDistance.ts
    var EARTH_RADIUS_METERS = 6371e3;
    var convertRadius = (meters, toUnit) => {
      const conversions = {
        meters: 1,
        kilometers: 1 / 1e3,
        miles: 1 / 1609.344,
        yards: 1.09361
      };
      return meters * conversions[toUnit];
    };
    function haversineDistance(lat1, lon1, lat2, lon2, unit = "meters") {
      if (lat1 === void 0 || lat1 === null || lon1 === void 0 || lon1 === null || lat2 === void 0 || lat2 === null || lon2 === void 0 || lon2 === null) {
        throw new TypeError("Latitude and longitude must be provided");
      }
      if (lat1 === lat2 && lon1 === lon2) {
        return 0;
      }
      if (typeof lat1 !== "number" || typeof lon1 !== "number" || typeof lat2 !== "number" || typeof lon2 !== "number") {
        throw new TypeError("Latitude and longitude must be numbers");
      }
      const validUnits = ["meters", "kilometers", "miles", "yards"];
      if (typeof unit !== "string" || !validUnits.includes(unit)) {
        throw new TypeError(
          'Unit must be one of "meters", "kilometers", "miles", or "yards"'
        );
      }
      const R = convertRadius(EARTH_RADIUS_METERS, unit);
      const lat1Rad = degreesToRadians(lat1);
      const lat2Rad = degreesToRadians(lat2);
      const deltaLat = degreesToRadians(lat2 - lat1);
      const deltaLon = degreesToRadians(lon2 - lon1);
      const a = Math.sin(deltaLon / 2) * Math.sin(deltaLon / 2) + Math.cos(lat1Rad) * Math.cos(lat2Rad) * Math.sin(deltaLat / 2) * Math.sin(deltaLat / 2);
      const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
      return R * c;
    }
    export {
      haversineDistance
    };
